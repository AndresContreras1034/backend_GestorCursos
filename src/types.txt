===== database/init.sql =====
CREATE TABLE IF NOT EXISTS usuarios (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    nombre TEXT NOT NULL,
    email TEXT NOT NULL UNIQUE,
    password TEXT NOT NULL,
    rol TEXT NOT NULL CHECK (rol IN ('estudiante', 'docente')),
    cedula TEXT UNIQUE,
    fecha_nacimiento TEXT,
    foto_perfil TEXT,
    fecha_creacion TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP,
    fecha_actualizacion TEXT,
    ultimo_login TEXT,
    activo INTEGER NOT NULL DEFAULT 1
);

CREATE TABLE IF NOT EXISTS cursos (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    nombre TEXT NOT NULL,
    descripcion TEXT NOT NULL,
    docente_id INTEGER NOT NULL,
    imagen_curso TEXT,
    fecha_creacion TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP,
    fecha_actualizacion TEXT,
    activo INTEGER NOT NULL DEFAULT 1,
    FOREIGN KEY (docente_id) REFERENCES usuarios(id)
);

CREATE TABLE IF NOT EXISTS inscripciones (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    estudiante_id INTEGER NOT NULL,
    curso_id INTEGER NOT NULL,
    fecha_inscripcion TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP,
    estado TEXT NOT NULL DEFAULT 'activa' CHECK (estado IN ('activa','retirada','completada')),
    FOREIGN KEY (estudiante_id) REFERENCES usuarios(id),
    FOREIGN KEY (curso_id) REFERENCES cursos(id),
    UNIQUE (estudiante_id, curso_id)
);

CREATE TABLE IF NOT EXISTS cronograma (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    curso_id INTEGER NOT NULL,
    titulo TEXT NOT NULL,
    descripcion TEXT,
    fecha_inicio TEXT,
    fecha_fin TEXT,
    orden INTEGER NOT NULL DEFAULT 0,
    fecha_creacion TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP,
    fecha_actualizacion TEXT,
    activo INTEGER NOT NULL DEFAULT 1,
    FOREIGN KEY (curso_id) REFERENCES cursos(id)
);

===== services/auth.service.js =====
const bcrypt = require('bcrypt');
const usuarioRepository = require('../repository/usuario.repository');

async function registrar(datos) {
  const { nombre, email, password, rol } = datos;

  if (!nombre || !email || !password || !rol) {
    throw new Error('Todos los campos son obligatorios');
  }

  const existente = usuarioRepository.buscarPorEmail(email);
  if (existente) {
    throw new Error('El correo ya está registrado');
  }

  const passwordHash = await bcrypt.hash(password, 10);

  const nuevoUsuario = usuarioRepository.crear({
    nombre,
    email,
    password: passwordHash,
    rol,
  });

  return nuevoUsuario;
}

async function login(email, password) {
  const usuario = usuarioRepository.buscarPorEmail(email);
  if (!usuario) {
    throw new Error('Credenciales inválidas');
  }

  const esValido = await bcrypt.compare(password, usuario.password);
  if (!esValido) {
    throw new Error('Credenciales inválidas');
  }

  return {
    id: usuario.id,
    nombre: usuario.nombre,
    email: usuario.email,
    rol: usuario.rol,
  };
}

module.exports = { registrar, login };

===== services/inscripcion.service.js =====
const inscripcionRepository = require('../repository/inscripcion.repository');
const cursoRepository = require('../repository/curso.repository');
const usuarioRepository = require('../repository/usuario.repository');

function inscribirse(datos) {
  const { estudianteId, cursoId } = datos;

  if (!estudianteId || !cursoId) {
    throw new Error('estudianteId y cursoId son obligatorios');
  }

  const estudiante = usuarioRepository.buscarPorId(estudianteId);
  if (!estudiante) {
    throw new Error('Estudiante no encontrado');
  }
  if (estudiante.rol !== 'estudiante') {
    throw new Error('Solo un usuario con rol estudiante puede inscribirse');
  }

  const curso = cursoRepository.buscarPorId(cursoId);
  if (!curso || curso.activo !== 1) {
    throw new Error('Curso no encontrado');
  }

  const existente = inscripcionRepository.buscarPorEstudianteYCurso(estudianteId, cursoId);
  if (existente) {
    throw new Error('El estudiante ya está inscrito en este curso');
  }

  return inscripcionRepository.crear({ estudianteId, cursoId });
}

function listarPorEstudiante(estudianteId) {
  if (!estudianteId) {
    throw new Error('estudianteId es obligatorio');
  }
  return inscripcionRepository.listarPorEstudiante(estudianteId);
}

module.exports = { inscribirse, listarPorEstudiante };

===== repository/inscripcion.repository.js =====
const db = require('../database/connection');

function crear({ estudianteId, cursoId }) {
  const stmt = db.prepare(`
    INSERT INTO inscripciones (estudiante_id, curso_id)
    VALUES (?, ?)
  `);
  const resultado = stmt.run(estudianteId, cursoId);
  return buscarPorId(resultado.lastInsertRowid);
}

function buscarPorId(id) {
  const stmt = db.prepare('SELECT * FROM inscripciones WHERE id = ?');
  return stmt.get(id);
}

function buscarPorEstudianteYCurso(estudianteId, cursoId) {
  const stmt = db.prepare(`
    SELECT * FROM inscripciones WHERE estudiante_id = ? AND curso_id = ? AND estado = 'activa'
  `);
  return stmt.get(estudianteId, cursoId);
}

function listarPorEstudiante(estudianteId) {
  const stmt = db.prepare(`
    SELECT i.id, i.curso_id AS cursoId, c.nombre, c.descripcion, c.imagen_curso AS imagenCurso,
           i.fecha_inscripcion AS fechaInscripcion, i.estado
    FROM inscripciones i
    JOIN cursos c ON i.curso_id = c.id
    WHERE i.estudiante_id = ?
  `);
  return stmt.all(estudianteId);
}

module.exports = { crear, buscarPorId, buscarPorEstudianteYCurso, listarPorEstudiante };

===== repository/curso.repository.js =====
const db = require('../database/connection');

function crear({ nombre, descripcion, docenteId }) {
  const stmt = db.prepare(`
    INSERT INTO cursos (nombre, descripcion, docente_id)
    VALUES (?, ?, ?)
  `);
  const resultado = stmt.run(nombre, descripcion, docenteId);

  return buscarPorId(resultado.lastInsertRowid);
}

function buscarPorId(id) {
  const stmt = db.prepare('SELECT * FROM cursos WHERE id = ?');
  return stmt.get(id);
}

function listarTodos(busqueda) {
  let query = `
    SELECT c.id, c.nombre, c.descripcion, c.docente_id AS docenteId, c.imagen_curso AS imagenCurso, u.nombre AS docenteNombre
    FROM cursos c
    JOIN usuarios u ON c.docente_id = u.id
    WHERE c.activo = 1
  `;
  const params = [];

  if (busqueda) {
    query += ` AND (c.nombre LIKE ? OR c.descripcion LIKE ? OR u.nombre LIKE ?)`;
    const like = `%${busqueda}%`;
    params.push(like, like, like);
  }

  const stmt = db.prepare(query);
  return stmt.all(...params);
}

const COLUMNAS_ACTUALIZABLES = {
  nombre: 'nombre',
  descripcion: 'descripcion',
  imagenCurso: 'imagen_curso',
};

function actualizar(id, datos) {
  const campos = [];
  const valores = [];

  for (const [campoJs, columnaDb] of Object.entries(COLUMNAS_ACTUALIZABLES)) {
    if (datos[campoJs] !== undefined) {
      campos.push(`${columnaDb} = ?`);
      valores.push(datos[campoJs]);
    }
  }

  campos.push('fecha_actualizacion = CURRENT_TIMESTAMP');

  const stmt = db.prepare(`
    UPDATE cursos
    SET ${campos.join(', ')}
    WHERE id = ?
  `);
  stmt.run(...valores, id);

  return buscarPorId(id);
}

module.exports = { crear, buscarPorId, listarTodos, actualizar };

===== services/curso.service.js =====
const cursoRepository = require('../repository/curso.repository');

function crearCurso(datos) {
  const { nombre, descripcion, docenteId } = datos;

  if (!nombre || !descripcion || !docenteId) {
    throw new Error('Nombre, descripción y docente son obligatorios');
  }

  const nuevoCurso = cursoRepository.crear({
    nombre,
    descripcion,
    docenteId,
  });

  return nuevoCurso;
}

function listarCursos(busqueda) {
  return cursoRepository.listarTodos(busqueda);
}

function actualizarCurso(id, datos) {
  const { nombre, descripcion, imagenCurso, docenteId } = datos;

  if (!nombre && !descripcion && !imagenCurso) {
    throw new Error('Debe enviar al menos un campo para actualizar');
  }

  const curso = cursoRepository.buscarPorId(id);
  if (!curso) {
    throw new Error('Curso no encontrado');
  }

  if (!docenteId) {
    throw new Error('docenteId es obligatorio para validar el curso');
  }
  if (curso.docente_id !== Number(docenteId)) {
    throw new Error('No tienes permiso para editar este curso');
  }

  if (imagenCurso && !imagenCurso.startsWith('data:image')) {
    throw new Error('La imagen del curso debe estar en formato base64 (data:image/...)');
  }

  const actualizado = cursoRepository.actualizar(id, {
    nombre,
    descripcion,
    imagenCurso,
  });

  return actualizado;
}

module.exports = { crearCurso, listarCursos, actualizarCurso };
